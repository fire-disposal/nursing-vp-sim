# 批次三：快照固化（case_data + rubric） 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 训练开始瞬间冻结病例内容（case_data）与评分标准（rubric，含护理记录动态维度），后续聊天/评分/回顾全部使用快照而非实时读取。教师改病例/改 rubric 不再影响进行中或已完成训练。
**Architecture:** 
- `training_records` 新增 `case_snapshot` JSONB 列
- `_create_record` 中同时写入 `case_snapshot`、`rubric_snapshot`、`prompt_snapshot`（一次性固化）
- 护理记录动态维度追加逻辑从 `score_engine.py` 抽出为纯函数 `build_final_rubric` → 创建时调用 + 评分 fallback 调用
- `score_engine.evaluate_training` 删除动态追加段，直接用 `record.rubric_snapshot`
- 所有 `case_data` 读取点改为 `record.case_snapshot or case.case_data` fallback
**Tech Stack:** Python/FastAPI/SQLAlchemy + PostgreSQL + Alembic + React/TypeScript
**Spec:** docs/superpowers/specs/2026-07-16-prototype-consolidation-design.md（批次三）
**Pre-requisite:** 批次一已完成（评分校验重构、rubric 相关代码可能有小变化）。执行前需 `git rebase` 对齐批次一的最新 commit。

```python
def build_final_rubric(base_rubric: dict, features: dict | None = None) -> dict:
    """返回最终评分 rubric（深拷贝，不修改入参）。

    Args:
        base_rubric: profile.rubric 或 load_rubric() 返回的基准 rubric dict
        features: resolve_features 后的能力开关 dict（检查 nursing_record 键）

    Returns:
        深拷贝后的 rubric dict；若 nursing_record 开启则追加护理记录维度并调高 raw_max
    """
    rubric = deepcopy(base_rubric)

    if features and features.get("nursing_record"):
        rubric.setdefault("dimensions", []).append(deepcopy(_NURSING_RECORD_DIMENSION))
        rubric["raw_max"] = rubric.get("raw_max", 57) + 15

    return rubric
```

```python
class TestCaseSnapshotIsolation:
    def test_detail_uses_snapshot_after_case_modified(self, client, student, test_case, db_session):
        """训练回顾使用快照数据，不受后续 case_data 修改影响"""
        _user, token = student

        # 设置原始病例数据
        test_case.case_data = deepcopy(ORIGINAL_CASE_DATA)
        db_session.commit()

        # 开始训练
        resp = client.post(
            "/api/training/start",
            json={"case_id": test_case.id},
            headers={"Authorization": f"Bearer {token}"},
        )
        assert resp.status_code == 200
        record_id = resp.json()["record_id"]

        # 修改 case_data
        test_case.case_data = deepcopy(MODIFIED_CASE_DATA)
        db_session.commit()

        # 获取详情 —— 应使用快照（原始数据）
        detail_resp = client.get(
            f"/api/training/records/{record_id}",
            headers={"Authorization": f"Bearer {token}"},
        )
        assert detail_resp.status_code == 200
        detail = detail_resp.json()

        assert detail["patient_info"]["name"] == "张大妈"
        assert detail["patient_info"]["age"] == 62
        assert detail["chief_complaint"] == "胸闷三天"
        # case_data 整体也应来自快照
        assert detail["case_data"]["patient_info"]["name"] == "张大妈"

class TestRubricSnapshot:
    def test_nursing_record_enabled_stores_full_rubric(self, client, student, test_case, db_session):
        """开启 nursing_record 时，rubric_snapshot 已含护理维度"""
        from models import TrainingRecord

        _user, token = student

        case_data = deepcopy(ORIGINAL_CASE_DATA)
        case_data["capabilities"] = {"nursing_record": True}
        test_case.case_data = case_data
        db_session.commit()

        resp = client.post(
            "/api/training/start",
            json={"case_id": test_case.id},
            headers={"Authorization": f"Bearer {token}"},
        )
        record_id = resp.json()["record_id"]

        record = db_session.query(TrainingRecord).filter(TrainingRecord.id == record_id).first()
        assert record.rubric_snapshot is not None
        assert record.case_snapshot is not None

        dims = record.rubric_snapshot.get("dimensions", [])
        dim_ids = [d["id"] for d in dims]
        assert "nursing_record" in dim_ids, f"期望含 nursing_record 维度，实际维度: {dim_ids}"
        assert record.rubric_snapshot.get("raw_max") > 57

    def test_nursing_record_disabled_stores_base_rubric(self, client, student, test_case, db_session):
        """未开启 nursing_record 时，rubric_snapshot 不含护理维度"""
        from models import TrainingRecord

        _user, token = student

        case_data = deepcopy(ORIGINAL_CASE_DATA)
        case_data["capabilities"] = {"nursing_record": False}
        test_case.case_data = case_data
        db_session.commit()

        resp = client.post(
            "/api/training/start",
            json={"case_id": test_case.id},
            headers={"Authorization": f"Bearer {token}"},
        )
        record_id = resp.json()["record_id"]

        record = db_session.query(TrainingRecord).filter(TrainingRecord.id == record_id).first()
        dims = record.rubric_snapshot.get("dimensions", [])
        dim_ids = [d["id"] for d in dims]
        assert "nursing_record" not in dim_ids
        assert record.rubric_snapshot.get("raw_max") == 57

    def test_retry_scoring_rubric_matches_first(self, client, student, test_case, db_session):
        """retry 评分与首评使用相同 rubric（均来自 snapshot）"""
        from models import TrainingRecord

        _user, token = student

        case_data = deepcopy(ORIGINAL_CASE_DATA)
        case_data["capabilities"] = {"nursing_record": True}
        test_case.case_data = case_data
        db_session.commit()

        resp = client.post(
            "/api/training/start",
            json={"case_id": test_case.id},
            headers={"Authorization": f"Bearer {token}"},
        )
        record_id = resp.json()["record_id"]

        record = db_session.query(TrainingRecord).filter(TrainingRecord.id == record_id).first()
        original_rubric = deepcopy(record.rubric_snapshot)

        # 模拟修改 case 能力（应不影响已有 snapshot）
        test_case.case_data["capabilities"] = {}
        db_session.commit()

        # 验证 snapshot 未变
        db_session.refresh(record)
        assert record.rubric_snapshot == original_rubric
        assert record.case_snapshot["capabilities"] == {"nursing_record": True}
```
